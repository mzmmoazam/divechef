**Goal:** Normalize the existing `Device` Prisma model to match the spec's identity contract (use the RDBI-read serial number as the device identity, not the unstable BLE address), and add the four `/api/devices` REST endpoints that the mobile add-a-device flow (P1) calls.

**Architecture:** The schema already has a `Device` model from earlier scaffolding, but its identity field is `bleAddress` — Android's BLE peripheral identifier rotates (random/resolvable addresses), so it's a poor identity choice. Switch to `serialNumber` (NOT NULL, hex-encoded RDBI bytes from the mobile `getDeviceInfo()`) with `(userId, serialNumber)` uniqueness. Drop `bleAddress`. Add four endpoints under `app/api/devices/`. `POST /api/dives` validates `meta.deviceSerial` against the user's registered devices.

**Tech Stack:** Next.js (App Router), Prisma, Postgres (Neon in production), Vitest.

## Task 1: Normalize the `Device` schema

### Step 1: Edit the Prisma model

Open `apps/web/prisma/schema.prisma`. Replace the `Device` block:

```prisma
model Device {
  id            String    @id @default(cuid())
  userId        String
  user          User      @relation(fields: [userId], references: [id], onDelete: Cascade)
  model         String
  serialNumber  String
  friendlyName  String?
  scanName      String?
  firmwareVersion String?
  registeredAt  DateTime  @default(now())
  lastSyncAt    DateTime?

  @@unique([userId, serialNumber])
  @@index([userId])
}
```

Notes:
- `bleAddress` removed (unstable; not the canonical identity).
- `serialNumber` is `String` (NOT NULL) — the hex-encoded RDBI ID_SERIAL bytes from mobile `getDeviceInfo()`.
- `friendlyName` is `String?` (nullable; user can rename).
- `scanName` is the BLE-advertised GAP name observed during scan — stored for diagnostics / Sentry context.
- `firmwareVersion` captured at registration — useful when triaging bugs.
- `registeredAt` (was `createdAt`) — clearer name for the registration moment.
- `onDelete: Cascade` on the user relation — when a user is hard-deleted, their devices go too. Spec calls for `DELETE /api/devices/:id` to NOT cascade to dives, but user-cascade-to-devices is fine.

### Step 2: Generate the migration

```bash
cd apps/web
DATABASE_URL='postgresql://...local-or-test-db...' \
  npx prisma migrate dev --name normalize_device_identity --skip-seed
```

The auto-generated migration will:

- Drop the `bleAddress` column.
- Add `friendlyName`, `scanName`, `firmwareVersion`, `registeredAt` columns.
- Drop the old unique on `(userId, bleAddress)`.
- Add the new unique on `(userId, serialNumber)`.
- Make `serialNumber` NOT NULL (Prisma will require an `@updatedAt` or default value if there are existing rows — for development DBs we accept the implicit "no rows yet"; for production, no rows exist yet either since this hasn't shipped).

If the migration fails because there's existing test data, the safe move pre-launch is to wipe with `npx prisma migrate reset --force`.

### Step 2: Implement `apps/web/app/api/devices/route.ts`

```ts
import { NextRequest, NextResponse } from 'next/server';
import { prisma } from '@/lib/prisma';
import { getAuthenticatedUserId } from '@/lib/auth'; // existing helper; check actual name

const SUPPORTED_MODELS = new Set([
  'peregrine', 'perdix', 'perdix-ai', 'perdix-2',
  'petrel-2', 'petrel-3', 'teric', 'nerd-2', 'tern',
  'unknown-shearwater',
]);

const FRIENDLY_LABEL: Record<string, string> = {
  peregrine: 'Peregrine',
  perdix: 'Perdix',
  'perdix-ai': 'Perdix AI',
  'perdix-2': 'Perdix 2',
  'petrel-2': 'Petrel 2',
  'petrel-3': 'Petrel 3',
  teric: 'Teric',
  'nerd-2': 'Nerd 2',
  tern: 'Tern',
  'unknown-shearwater': 'Shearwater',
};

export async function GET(req: NextRequest) {
  const userId = await getAuthenticatedUserId(req);
  if (!userId) return NextResponse.json({ error: 'unauthenticated' }, { status: 401 });

  const devices = await prisma.device.findMany({
    where: { userId },
    orderBy: { registeredAt: 'asc' },
  });
  return NextResponse.json({ devices });
}

export async function POST(req: NextRequest) {
  const userId = await getAuthenticatedUserId(req);
  if (!userId) return NextResponse.json({ error: 'unauthenticated' }, { status: 401 });

  const body = await req.json();
  const { model, serialNumber, friendlyName, scanName, firmwareVersion } = body ?? {};

  if (typeof model !== 'string' || !SUPPORTED_MODELS.has(model)) {
    return NextResponse.json(
      { error: 'invalid_model', detail: `model must be one of: ${[...SUPPORTED_MODELS].join(', ')}` },
      { status: 400 }
    );
  }
  if (typeof serialNumber !== 'string' || serialNumber.length === 0) {
    return NextResponse.json(
      { error: 'invalid_serial', detail: 'serialNumber is required' },
      { status: 400 }
    );
  }

  // Upsert: idempotent on (userId, serialNumber).
  const device = await prisma.device.upsert({
    where: { userId_serialNumber: { userId, serialNumber } },
    create: {
      userId,
      model,
      serialNumber,
      friendlyName: friendlyName ?? FRIENDLY_LABEL[model] ?? null,
      scanName: scanName ?? null,
      firmwareVersion: firmwareVersion ?? null,
    },
    update: {
      // Only update soft fields on re-register (e.g., scanName / firmwareVersion seen freshly)
      scanName: scanName ?? undefined,
      firmwareVersion: firmwareVersion ?? undefined,
    },
  });

  return NextResponse.json({ device }, { status: 201 });
}
```

### Step 2: Implement `apps/web/app/api/devices/[id]/route.ts`

```ts
import { NextRequest, NextResponse } from 'next/server';
import { prisma } from '@/lib/prisma';
import { getAuthenticatedUserId } from '@/lib/auth';

export async function PATCH(
  req: NextRequest,
  { params }: { params: { id: string } }
) {
  const userId = await getAuthenticatedUserId(req);
  if (!userId) return NextResponse.json({ error: 'unauthenticated' }, { status: 401 });

  const body = await req.json();
  const { friendlyName } = body ?? {};
  if (typeof friendlyName !== 'string' || friendlyName.length === 0) {
    return NextResponse.json({ error: 'invalid_friendlyName' }, { status: 400 });
  }

  // Ensure the device belongs to the authenticated user.
  const existing = await prisma.device.findUnique({ where: { id: params.id } });
  if (!existing || existing.userId !== userId) {
    return NextResponse.json({ error: 'not_found' }, { status: 404 });
  }

  const device = await prisma.device.update({
    where: { id: params.id },
    data: { friendlyName },
  });
  return NextResponse.json({ device });
}

export async function DELETE(
  req: NextRequest,
  { params }: { params: { id: string } }
) {
  const userId = await getAuthenticatedUserId(req);
  if (!userId) return NextResponse.json({ error: 'unauthenticated' }, { status: 401 });

  const existing = await prisma.device.findUnique({ where: { id: params.id } });
  if (!existing || existing.userId !== userId) {
    return NextResponse.json({ error: 'not_found' }, { status: 404 });
  }

  // No cascade — dives keep their deviceSerial reference for history.
  // The Device row is removed from active inventory only.
  await prisma.device.delete({ where: { id: params.id } });
  return NextResponse.json({ ok: true });
}
```
